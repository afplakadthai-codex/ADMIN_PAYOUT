  `currency` char(3) NOT NULL DEFAULT 'USD',
  `status` enum('requested','approved','rejected','paid','cancelled') NOT NULL DEFAULT 'requested',

CREATE TABLE `seller_balances` (
  `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT,
  `seller_id` bigint(20) unsigned NOT NULL COMMENT 'users.id of the seller',
  `currency` char(3) NOT NULL DEFAULT 'USD',
  `pending_balance` decimal(14,4) NOT NULL DEFAULT 0.0000 COMMENT 'Earned but not yet cleared (awaiting order delivery/clearing period)',
  `available_balance` decimal(14,4) NOT NULL DEFAULT 0.0000 COMMENT 'Cleared and ready for payout request',
  `held_balance` decimal(14,4) NOT NULL DEFAULT 0.0000 COMMENT 'In an active payout request — awaiting admin payment',
  `paid_out_balance` decimal(14,4) NOT NULL DEFAULT 0.0000 COMMENT 'Cumulative total paid out to seller (never decremented)',
  `total_earned_gross` decimal(14,4) NOT NULL DEFAULT 0.0000 COMMENT 'Lifetime gross sales credited to this seller',
  `total_platform_fee` decimal(14,4) NOT NULL DEFAULT 0.0000 COMMENT 'Lifetime platform fees deducted',
  `created_at` datetime NOT NULL DEFAULT current_timestamp(),
  `updated_at` datetime NOT NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_seller_balances_seller` (`seller_id`),
  KEY `idx_sb_currency` (`currency`,`available_balance`)
) ENGINE=InnoDB AUTO_INCREMENT=47 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='Running balance snapshot per seller — reconcile against seller_ledger'

CREATE TABLE `seller_balance_entries` (

  `order_item_id` bigint(20) unsigned DEFAULT NULL,

  `risk_score` int(11) NOT NULL DEFAULT 0,
  `risk_flags_json` text DEFAULT NULL,
  `hold_reason` varchar(191) DEFAULT NULL,
  `release_at` datetime DEFAULT NULL,
  `available_at` datetime DEFAULT NULL,
  `paid_out_at` datetime DEFAULT NULL,
  `payout_id` bigint(20) unsigned DEFAULT NULL,
  `source` varchar(50) NOT NULL DEFAULT 'order_paid',
  `created_at` datetime NOT NULL DEFAULT current_timestamp(),
  `updated_at` datetime NOT NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  PRIMARY KEY (`id`),
  UNIQUE KEY `uniq_seller_balance_entry_item` (`seller_id`,`order_id`,`order_item_id`),
  KEY `idx_seller_balance_entries_status_release` (`status`,`release_at`),
  KEY `idx_seller_balance_entries_seller_status` (`seller_id`,`status`),
  KEY `idx_seller_balance_entries_order` (`order_id`),
  KEY `idx_seller_balance_entries_payout` (`payout_id`)
) ENGINE=InnoDB AUTO_INCREMENT=22 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
